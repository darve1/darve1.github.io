<%*
/*
  다른 글 링크 넣기 (Templater) — 글을 연 상태에서 명령 팔레트: "Templater: Insert 글 링크" (Ctrl+Alt+L)
  _posts 의 글을 제목으로 골라 [제목](_posts/…md) 링크를 넣는다.
  블로그에서는 _plugins/obsidian-content.rb 가 /posts/슬러그/ 주소로 바꾸고,
  아직 발행하지 않은 글이면 링크 없이 글자만 보인다.
*/
if (tp.config.run_mode !== 1) {
  throw new Error("글을 연 상태에서 'Templater: Insert 글 링크' 명령으로 실행하세요.");
}

const current = tp.config.target_file.path;
const posts = app.vault
  .getMarkdownFiles()
  .filter((f) => f.path.startsWith("_posts/") && f.path !== current)
  .sort((a, b) => b.basename.localeCompare(a.basename));

// front matter 의 제목과 발행 여부 (published 가 없으면 Jekyll처럼 발행으로 본다)
const info = (f) => {
  const fm = app.metadataCache.getFileCache(f)?.frontmatter ?? {};
  return { title: String(fm.title || f.basename), published: fm.published !== false };
};

if (posts.length === 0) {
  new tp.obsidian.Notice("링크할 다른 글이 없습니다.");
} else {
  const labels = posts.map((f) => {
    const { title, published } = info(f);
    return (published ? "" : "[비공개] ") + title + " · " + f.basename.slice(0, 10);
  });
  const file = await tp.system.suggester(labels, posts, true, "링크할 글");
  const { title } = info(file);
  const text = (await tp.system.prompt("링크에 보일 글자", title, true)).trim() || title;
  tR += app.fileManager.generateMarkdownLink(file, current, undefined, text);
}
-%>
