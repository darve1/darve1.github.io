<%*
/*
  블로그 새 글 (Templater) — 명령 팔레트: "Templater: Create 새 글" (Ctrl+Alt+N)
  제목·카테고리·소분류·슬러그·요약을 묻고 _posts/YYYY-MM-DD-슬러그.md 를 만든다.
  어느 질문에서든 Esc로 취소하면 Templater가 만들던 노트를 지운다.
*/
if (tp.config.run_mode !== 0) {
  throw new Error("'Templater: Create 새 글' 명령으로 실행하세요.");
}

const CATEGORIES = ["개발", "CTF/Wargame", "Bug Bounty", "블로그/기술문서", "논문/컨퍼런스", "공모전/자격증"];

const title = (await tp.system.prompt("글 제목", "", true)).trim();
if (!title) {
  throw new Error("제목이 비어 있습니다.");
}
const category = await tp.system.suggester(CATEGORIES, CATEGORIES, true, "카테고리");
const sub = (await tp.system.prompt("소분류 (선택: 없으면 비워 두고 Enter)", "", true)).trim();
const slugInput = await tp.system.prompt("주소용 영문 슬러그 (예: dreamhack-xss)", "", true);
const slug = slugInput.toLowerCase().trim().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "") || tp.date.now("HHmm");
const description = (await tp.system.prompt("한 줄 요약 (선택: 홈 목록에 보임)", "", true)).trim();

await tp.file.move("_posts/" + tp.date.now("YYYY-MM-DD") + "-" + slug);

// JSON 문자열은 그대로 YAML로 읽힌다 (따옴표·콜론이 들어간 제목도 안전)
const fm = ["---", "title: " + JSON.stringify(title)];
if (description) {
  fm.push("description: " + JSON.stringify(description));
}
fm.push(
  "categories: " + JSON.stringify(sub ? [category, sub] : [category]),
  "tags: []",
  "published: false",
  "---",
  "",
  "## 개요",
  "",
  ""
);
tR += fm.join("\n");
-%>
<% tp.file.cursor() %>
