<%*
/*
  블로그 발행 (Templater) — 글을 연 상태에서 명령 팔레트: "Templater: Insert 발행" (Ctrl+Alt+P)
  published 를 체크하고 Obsidian Git "Commit-and-sync"로 커밋·푸시한다.
  이미 발행한 글을 고친 뒤 다시 실행하면 수정 사항만 올라간다.
*/
if (tp.config.run_mode !== 1) {
  throw new Error("발행할 글을 연 상태에서 'Templater: Insert 발행' 명령으로 실행하세요.");
}

const file = tp.config.target_file;
const isPost = file.path.startsWith("_posts/") && /^\d{4}-\d{2}-\d{2}-.+\.md$/.test(file.name);

if (!isPost) {
  new tp.obsidian.Notice("발행할 글(_posts/YYYY-MM-DD-슬러그.md)을 연 상태에서 실행하세요.");
} else {
  // 템플릿 삽입(빈 내용)이 끝난 뒤에 속성을 바꾸고 커밋·푸시한다
  tp.hooks.on_all_templates_executed(async () => {
    await app.fileManager.processFrontMatter(file, (fm) => {
      fm.published = true;
    });
    new tp.obsidian.Notice("발행: " + file.basename + " → 커밋·푸시합니다");
    await app.commands.executeCommandById("obsidian-git:push");
  });
}
-%>
