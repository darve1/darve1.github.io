<%*
/*
  글 썸네일 정하기 (Templater) — 글을 연 상태에서 명령 팔레트: "Templater: Insert 썸네일" (Ctrl+Alt+T)
  이 글의 첨부 폴더(assets/posts/<글 파일 이름>/)에 있는 이미지 중 하나를 골라 front matter 의 image 로 넣는다.
  홈 글 목록 카드 오른쪽과 글 맨 위에 보인다. 이미 정한 썸네일을 빼는 것도 여기서 고른다.
*/
if (tp.config.run_mode !== 1) {
  throw new Error("글을 연 상태에서 'Templater: Insert 썸네일' 명령으로 실행하세요.");
}

const file = tp.config.target_file;
const isPost = file.path.startsWith("_posts/") && /^\d{4}-\d{2}-\d{2}-.+\.md$/.test(file.name);

if (!isPost) {
  new tp.obsidian.Notice("썸네일을 정할 글(_posts/YYYY-MM-DD-슬러그.md)을 연 상태에서 실행하세요.");
} else {
  const folder = "assets/posts/" + file.basename + "/";
  const images = app.vault.getFiles()
    .filter((f) => f.path.startsWith(folder) && ["png", "jpg", "jpeg", "gif", "webp", "svg"].includes(f.extension.toLowerCase()))
    .sort((a, b) => a.name.localeCompare(b.name));
  const current = app.metadataCache.getFileCache(file)?.frontmatter?.image;

  const REMOVE = "(썸네일 빼기)";
  const choices = images.map((f) => f.name);
  if (current) {
    choices.push(REMOVE);
  }

  if (choices.length === 0) {
    new tp.obsidian.Notice("이 글의 첨부 폴더에 이미지가 없습니다. 먼저 이미지를 붙여넣거나 Ctrl+Alt+F로 넣으세요.");
  } else {
    const picked = await tp.system.suggester(choices, choices, true, "썸네일로 쓸 이미지 (1200x630 비율 권장)");
    tp.hooks.on_all_templates_executed(async () => {
      await app.fileManager.processFrontMatter(file, (fm) => {
        if (picked === REMOVE) {
          delete fm.image;
        } else {
          fm.image = "/" + folder + picked;
        }
      });
      new tp.obsidian.Notice(picked === REMOVE ? "썸네일을 뺐습니다" : "썸네일: " + picked);
    });
  }
}
-%>
