<%*
/*
  YouTube 영상 넣기 (Templater) — 글을 연 상태에서 명령 팔레트: "Templater: Insert 유튜브" (Ctrl+Alt+Y)
  커서 위치에 ![설명](YouTube 주소)를 넣는다. Obsidian에서는 바로 재생되고,
  블로그에서는 _plugins/obsidian-content.rb 가 플레이어로 바꾼다. (&t=90 이면 90초부터 재생)
*/
if (tp.config.run_mode !== 1) {
  throw new Error("글을 연 상태에서 'Templater: Insert 유튜브' 명령으로 실행하세요.");
}

const url = (await tp.system.prompt("YouTube 주소", "", true)).trim();
const isYoutube = /^https?:\/\/(www\.|m\.)?(youtube\.com\/(watch\?|shorts\/|embed\/|live\/)|youtu\.be\/)/.test(url);

if (!isYoutube) {
  new tp.obsidian.Notice("YouTube 주소가 아닙니다: " + url);
} else {
  const caption = (await tp.system.prompt("영상 아래 설명 (선택: 없으면 비워 두고 Enter)", "", true)).trim();
  tR += "![" + caption + "](" + url + ")\n";
}
-%>
